from sklearn.metrics import (
    classification_report,
    confusion_matrix,
    accuracy_score,
    balanced_accuracy_score
)

def evaluate_model(name, y_true, predictions):

    cm = confusion_matrix(
        y_true,
        predictions,
        labels=[0, 1, 2, 3]
    )

    print("\n", name)
    print(classification_report(
